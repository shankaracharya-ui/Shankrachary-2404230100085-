SELECT s.student_name, s.city, c.course_name, m.subject, m.marks
FROM students s
JOIN courses c ON s.course_id = c.course_id
LEFT JOIN results m ON s.student_id = m.student_id;

ANSWER - 14
SELECT c.course_name,
       COUNT(s.student_id) AS total_students,
       AVG(m.marks) AS avg_marks
FROM courses c
JOIN students s ON c.course_id = s.course_id
JOIN results m ON s.student_id = m.student_id
GROUP BY c.course_name
HAVING AVG(m.marks) > 90;

ANSWER - 15
WITH Student_Avg AS (
SELECT s.student_id, s.student_name, s.course_id,
AVG(r.marks) AS avg_marks
FROM students s
JOIN results r ON s.student_id = r.student_id
GROUP BY s.student_id, s.student_name, s.course_id
)
SELECT student_name,
avg_marks,
FROM Student_Avg;
